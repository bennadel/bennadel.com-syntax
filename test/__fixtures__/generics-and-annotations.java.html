package com.bennadel.archive; import java.util.List; import java.util.stream.Collectors; @Service public class PostIndexer<T extends Comparable<T>> { private static final int BATCH = 500; @Autowired private Repository repository; /** * Reindex everything newer than the given id. */ public List<String> reindex(long sinceId) throws IndexException { return repository .findAllNewerThan(sinceId, BATCH) .stream() .map(post -> post.getTitle().trim()) .filter(title -> !title.isEmpty()) .collect(Collectors.toList()); } }
